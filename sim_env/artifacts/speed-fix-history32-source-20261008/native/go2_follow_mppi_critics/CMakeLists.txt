cmake_minimum_required(VERSION 3.16)
project(go2_follow_mppi_critics)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(xtensor REQUIRED)
find_package(xsimd REQUIRED)
# 与Humble MPPI的张量配置保持一致，避免插件和宿主使用不同的分配器ABI。
add_library(go2_follow_mppi_critics SHARED src/follow_speed_critic.cpp src/braking_critic.cpp src/braking_api.cpp)
target_compile_features(go2_follow_mppi_critics PUBLIC cxx_std_17)
target_compile_definitions(go2_follow_mppi_critics PUBLIC XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
target_link_libraries(go2_follow_mppi_critics xtensor xtensor::optimize xtensor::use_xsimd)
ament_target_dependencies(go2_follow_mppi_critics nav2_mppi_controller rclcpp geometry_msgs nav_msgs std_msgs pluginlib)
pluginlib_export_plugin_description_file(nav2_mppi_controller critics.xml)
install(TARGETS go2_follow_mppi_critics LIBRARY DESTINATION lib)
ament_package()
